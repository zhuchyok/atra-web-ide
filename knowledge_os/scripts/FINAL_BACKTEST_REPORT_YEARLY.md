**РЕШЕНИЕ А:** Добавить **подтверждение сильного тренда на H4**

```python
def get_h4_trend_strength(symbol):
    """Получить силу тренда на H4"""
    df_h4 = get_data(symbol, '4h')

    # EMA расстояние
    ema_distance = abs(df_h4['ema_fast'] - df_h4['ema_slow']) / df_h4['ema_slow']

    # MACD импульс
    macd_strength = abs(df_h4['macd_hist']) / abs(df_h4['macd'])

    # ADX (направленность движения)
    adx = calculate_adx(df_h4, period=14)

    trend_strength = (ema_distance + macd_strength + adx/100) / 3

    return trend_strength

# В фильтрах добавить:
if get_h4_trend_strength(symbol) < 0.02:  # минимум 2% сила тренда
    reject_signal("H4 тренд слишком слабый")
```

    """Проверить, сейчас часы высокой ликвидности"""
    hour_utc = datetime.utcnow().hour

    # Высокая активность: 8:00-16:00 UTC (Европа + США)
    if 8 <= hour_utc <= 16:
        return True

    return False
