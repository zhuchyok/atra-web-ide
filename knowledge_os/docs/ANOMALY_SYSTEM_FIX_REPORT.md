def get_anomaly_emoji(score: float) -> str:

        return "🔥"
    elif score >= 0.6:  # 3 кружка - высокая аномалия
        return "⚠️"
    elif score >= 0.2:  # 1-2 кружка - низкая аномалия
        return "📈"
    else:  # 0 кружков - нормальная активность
        return "✅"
