```python
async def set_trade_mode_cmd(update, context):
    # ... инициализация ...

    args = context.args
    if not args:
        current_mode = user_data.get("trade_mode", "spot")
        current_mode_emoji = "💵" if current_mode == "spot" else "⚡"

        msg = (
            f"🔄 <b>Управление режимом торговли</b>\n\n"
            f"Текущий режим: {current_mode_emoji} {current_mode.upper()}\n\n"
            f"💵 <b>SPOT режим:</b>\n"
            f"• Только LONG позиции\n"
            f"• Без плеча (1x)\n"
            f"• Более безопасная торговля\n"
            f"• Рекомендуется для начинающих\n\n"
            f"⚡ <b>FUTURES режим:</b>\n"
            f"• LONG и SHORT позиции\n"
            f"• Динамическое плечо (0.5x-5x)\n"
            f"• Более активная торговля\n"
            f"• Рекомендуется для опытных\n\n"
            f"Выберите режим кнопкой ниже:"
        )

        keyboard = InlineKeyboardMarkup([
            [
                InlineKeyboardButton("💵 SPOT", callback_data="trade_mode_spot"),
                InlineKeyboardButton("⚡ FUTURES", callback_data="trade_mode_futures")
            ]
        ])

        await update.message.reply_text(msg, parse_mode="HTML", reply_markup=keyboard)
        return
```
