```python
# Быстрое переключение между источниками
for source in sorted(sources, key=lambda x: x.priority):
    if not source.enabled:
        continue

    try:
        data = await self._fetch_from_source(source, symbol, base_symbol, data_type, **kwargs)
        if data:
            # Кэшируем результат
            self.cache[cache_key] = (data, asyncio.get_event_loop().time())
            logging.info(f"✅ Получены данные {data_type} для {symbol} из {source.name}")
            return data
    except Exception as e:
        logging.debug(f"❌ {source.name} недоступен: {e}")
        continue
```
