async def stop_telegram_bot():
    """Остановка Telegram бота"""
    global _application
    try:
        if _application:
            await _application.stop()
            await _application.shutdown()
            _application = None
            print("[TelegramBot] Бот остановлен")
    except Exception as e:
        print(f"[TelegramBot] Ошибка остановки бота: {e}")
