### ⚡ **1. Динамическое плечо (1-20x)**

**Алгоритм расчета:**

```python
def calculate_dynamic_leverage(self, row: pd.Series) -> float:
    base_leverage = self.base_leverage

    # Множитель на основе силы тренда
    trend_multiplier = min(2.0, 1.0 + row['trend_strength'] * 0.5)

    # Множитель на основе волатильности (ATR)
    atr_pct = row['atr'] / row['Close'] * 100
    volatility_multiplier = 1.0
    if atr_pct < 1.0:  # Низкая волатильность
        volatility_multiplier = 1.2
    elif atr_pct > 3.0:  # Высокая волатильность
        volatility_multiplier = 0.8

    # Множитель на основе RSI
    rsi_multiplier = 1.0
    if row['rsi'] < 30:  # Перепроданность
        rsi_multiplier = 1.1
    elif row['rsi'] > 70:  # Перекупленность
        rsi_multiplier = 0.9

    # Итоговое плечо
    dynamic_leverage = base_leverage * trend_multiplier * volatility_multiplier * rsi_multiplier

    # Ограничиваем от 1 до 20
    return max(1.0, min(dynamic_leverage, 20.0))
```

**Логика:**

- **Сильный тренд** → увеличиваем плечо
- **Низкая волатильность** → увеличиваем плечо
- **Перепроданность/перекупленность** → корректируем плечо
- **Ограничения:** минимум 1x, максимум 20x

```python
def calculate_dynamic_risk(self, row: pd.Series) -> float:
    base_risk = self.base_risk

    # Корректировка на основе волатильности
    atr_pct = row['atr'] / row['Close'] * 100
    volatility_adjustment = 1.0
    if atr_pct < 1.0:
        volatility_adjustment = 1.2  # Увеличиваем риск при низкой волатильности
    elif atr_pct > 3.0:
        volatility_adjustment = 0.8  # Уменьшаем риск при высокой волатильности

    # Корректировка на основе объема
    volume_adjustment = 1.0
    if row['volume_ratio'] > 1.5:
        volume_adjustment = 1.1  # Увеличиваем риск при высоком объеме
    elif row['volume_ratio'] < 0.5:
        volume_adjustment = 0.9  # Уменьшаем риск при низком объеме

    # Корректировка на основе силы тренда
    trend_adjustment = 1.0
    if row['trend_strength'] > 2.0:
        trend_adjustment = 1.1  # Увеличиваем риск при сильном тренде
    elif row['trend_strength'] < 0.5:
        trend_adjustment = 0.9  # Уменьшаем риск при слабом тренде

    # Итоговый риск
    dynamic_risk = base_risk * volatility_adjustment * volume_adjustment * trend_adjustment

    # Ограничиваем от 0.5 до 5%
    return max(0.5, min(dynamic_risk, 5.0))
```

**Логика:**

- **Низкая волатильность** → увеличиваем риск
- **Высокий объем** → увеличиваем риск
- **Сильный тренд** → увеличиваем риск
- **Ограничения:** минимум 0.5%, максимум 5%

### 🎯 **3. Динамические тейк-профиты (1-10%)**

**Алгоритм расчета:**

```python
def calculate_dynamic_tp_levels(self, row: pd.Series, side: str) -> Tuple[float, float]:
    atr = row['atr']
    price = row['Close']

    # Базовые множители ATR
    tp1_mult = 2.0
    tp2_mult = 4.0

    # Корректировка на основе ширины Bollinger Bands
    bb_width = row['bb_width']
    if bb_width > 10:  # Широкие полосы - увеличиваем TP
        tp1_mult *= 1.2
        tp2_mult *= 1.2
    elif bb_width < 5:  # Узкие полосы - уменьшаем TP
        tp1_mult *= 0.8
        tp2_mult *= 0.8

    # Корректировка на основе силы тренда
    trend_strength = row['trend_strength']
    if trend_strength > 2.0:  # Сильный тренд - увеличиваем TP
        tp1_mult *= 1.3
        tp2_mult *= 1.3
    elif trend_strength < 0.5:  # Слабый тренд - уменьшаем TP
        tp1_mult *= 0.7
        tp2_mult *= 0.7

    # Рассчитываем уровни TP
    if side == 'long':
        tp1 = price + atr * tp1_mult
        tp2 = price + atr * tp2_mult
    else:
        tp1 = price - atr * tp1_mult
        tp2 = price - atr * tp2_mult

    return tp1, tp2
```

**Логика:**

- **Широкие полосы Боллинджера** → увеличиваем TP
- **Сильный тренд** → увеличиваем TP
- **Узкие полосы** → уменьшаем TP
- **Слабый тренд** → уменьшаем TP

### 🔄 **4. Система DCA (Dollar Cost Averaging)**

**Условия для DCA:**

```python
def should_dca(self, position: Dict, row: pd.Series) -> bool:
    # Проверяем, что прошло достаточно времени с последнего DCA
    if position['dca_levels']:
        last_dca_time = position['dca_levels'][-1]['time']
        time_diff = (row.name - last_dca_time).total_seconds() / 3600  # часы
        if time_diff < 4:  # Минимум 4 часа между DCA
            return False

    # Проверяем объем и RSI
    volume_ok = row['volume_ratio'] > 1.0
    rsi_ok = (position['side'] == 'long' and row['rsi'] < 40) or \
             (position['side'] == 'short' and row['rsi'] > 60)

    return volume_ok and rsi_ok
```

**Логика DCA:**

- **Цена движется против позиции** (2% для лонга, -2% для шорта)
- **Минимум 4 часа** между DCA
- **Высокий объем** (>1.0 от среднего)
- **Подходящий RSI** (<40 для лонга, >60 для шорта)
- **Увеличиваем позицию на 50%** от текущего размера

```python
def generate_signals(self) -> pd.DataFrame:
    df = self.df.copy()

    # Сигналы на основе EMA
    df['long_signal'] = (df['ema7'] > df['ema25']) & (df['ema25'] > df['ema50'])
    df['short_signal'] = (df['ema7'] < df['ema25']) & (df['ema25'] < df['ema50'])

    # Фильтры
    df['volume_filter'] = df['volume_ratio'] > 0.8
    df['rsi_filter'] = (df['rsi'] > 30) & (df['rsi'] < 70)
    df['bb_filter'] = (df['Close'] > df['bb_lower']) & (df['Close'] < df['bb_upper'])

    # Итоговые сигналы
    df['long_entry'] = df['long_signal'] & df['volume_filter'] & df['rsi_filter'] & df['bb_filter']
    df['short_entry'] = df['short_signal'] & df['volume_filter'] & df['rsi_filter'] & df['bb_filter']

    return df
```
